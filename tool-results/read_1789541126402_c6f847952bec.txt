     3→//
     4→// Implemented stages: Application (scaffold), FSOW Orientation (stage 1),
     5→// Automated Interview Assessment (stage 2), Responsibility & Reward
     6→// Agreement (stage 3 — persisted as the OFFER stage/statuses; the
     7→// candidate-facing label is "Agreement", see src/config/recruitment.ts),
     8→// the Provisional Appointment (stage 4 — lazy issuance, one letter per
     9→// candidate, see src/server/services/appointment.ts), Staff Activation
    10→// (stage 5 — explicit in-portal activation, one Staff identity per
    11→// candidate, see src/server/services/activation.ts), and Staff Onboarding
    12→// (stage 6 — STAFF-SIDE operational briefing after recruitment completes:
    13→// StaffOnboarding + StaffOnboardingStep, see src/server/services/
    14→// onboarding.ts). Onboarding deliberately adds NO CandidateStatus value
    15→// and NO RecruitmentStage value — APPROVED_STAFF remains the terminal
    16→// recruitment status, and StageEvent stays recruitment-only.

    19→// "notification_service") records transactional email intents for
    20→// lifecycle events via the central notification service
    21→// (src/server/notifications/) — outbox pattern, dedupe-key exactly-once,
    22→// Mailjet delivery AFTER the business transaction commits.
    23→// See docs/CONTENT-QUESTIONS.md before extending: stage rules are
    24→// finalized stage-by-stage in the product planning phase.

    79→/// Delivery status of a transactional notification.

   318→/// The candidate's issued Provisional Letter of Appointment — stage 4 of
   319→/// the journey. Issued lazily and exactly once, server-side, when an
   320→/// OFFER_ACCEPTED candidate first opens the appointment stage
   321→/// (src/server/services/appointment.ts); viewing again is a read-only
   322→/// record and NEVER re-issues or mutates the letter.
   323→///
   324→/// Hybrid historical-integrity model: the columns below are the immutable,
   325→/// candidate-specific SNAPSHOT (who, which letter, when, under which
   326→/// content version) while the letter TEXT itself lives in an append-only
   327→/// content registry (src/config/appointment.ts, APPOINTMENT_CONTENT_
   328→/// VERSIONS) resolved by `version`. An appointment issued under v1.0 must
   329→/// stay reproducible as v1.0 forever — later wording changes append new
   330→/// registry versions and never touch issued ones. There is deliberately no
   331→/// separate effectiveDate column: the letter date IS `issuedAt` for the
   332→/// MVP (docs/CONTENT-QUESTIONS.md §23 keeps the distinction open).
   333→model Appointment {
   334→  id          String    @id @default(cuid())
   335→  candidateId String    @unique
   336→  candidate   Candidate @relation(fields: [candidateId], references: [id], onDelete: Cascade)
   337→  /// Official letter reference in the FFL/HR/PLA/NNN format. Allocated
   338→  /// server-side from the AppointmentSequence inside the issuance
   339→  /// transaction — never derived from client input, never from count(*).
   340→  /// The unique constraint is the final backstop against duplicates.
   341→  appointmentReference String @unique
   342→  /// Pinned content-registry version (e.g. "1.0") the letter was issued
   343→  /// under. Immutable after issuance.
   344→  version     String
   345→  /// Printed candidate name, snapshotted from Candidate.fullName at
   346→  /// issuance. Never read from the client at render time.
   347→  printedName String
   348→  /// Official appointment designation, snapshotted at issuance. The
   349→  /// approved MVP value is "Remote Staff" (the designation used by the
   350→  /// official Provisional Letter of Appointment) — not renamed to any
   351→  /// other role title (docs/CONTENT-QUESTIONS.md §21).
   352→  roleTitle   String
   353→  /// Server issuance timestamp. Doubles as the printed letter date for
   354→  /// the MVP; no separate effectiveDate exists (§23).
   355→  issuedAt    DateTime
   356→  createdAt   DateTime  @default(now())
   357→  updatedAt   DateTime  @updatedAt
   358→
   359→  @@map("appointments")
   360→}

   382→  @@map("appointment_sequences")

   438→/// The Staff member's authenticated Staff Portal access (Stage 7). One row
   439→/// per Staff record, created in the SAME transaction as the Staff row by
   440→/// staff activation (and idempotently by the backfill script for pre-existing
   441→/// Staff). The account owns login codes and portal sessions and carries the
   442→/// future suspension flag — nothing else:
   443→///
   444→/// - It duplicates NO identity data. Name, email, phone and appointment
   445→///   facts are reached through StaffAccount → Staff → Candidate; the login
   446→///   email IS the candidate's registered email (brief §3/§4).
   447→/// - It stores NO credentials. Login is email OTP (hashed codes live on
   448→///   StaffLoginCode); sessions are DB rows with only the token hash
   449→///   (StaffSession).
   450→/// - status = SUSPENDED blocks code requests, verification and session
   451→///   resolution; there is deliberately no admin UI in this stage.
   452→model StaffAccount {
   453→  id      String             @id @default(cuid())
   454→  /// One portal account per Staff identity.
   455→  staffId String             @unique
   456→  staff   Staff              @relation(fields: [staffId], references: [id], onDelete: Cascade)
   457→  /// ACTIVE by default; SUSPENDED refuses every auth path (brief §17).
   458→  status  StaffAccountStatus @default(ACTIVE)
   459→  /// Set on each successful OTP verification.
   460→  lastLoginAt DateTime?
   461→  createdAt  DateTime       @default(now())
   462→  updatedAt  DateTime       @updatedAt
   463→
   464→  loginCodes StaffLoginCode[]
   465→  sessions   StaffSession[]
   466→
   467→  @@map("staff_accounts")
   468→}

   470→/// One emailed email-OTP login code for a Staff Portal account (Stage 7).
   471→///
   472→/// Security properties (brief §6/§7/§10): the 6-digit code is generated with
   473→/// cryptographically secure randomness and stored ONLY as a sha256 hash —
   474→/// the plaintext lives in memory just long enough to render the one-shot
   475→/// notification payload and is never persisted or logged. A code is
   476→/// single-use (consumedAt latch), expires after 10 minutes, tolerates at
   477→/// most 5 failed verification attempts, and only the LATEST code of an
   478→/// account is ever verifiable (a newer request replaces older ones).
   479→/// Request-side cooldown and rate limits are enforced by the auth service
   480→/// from createdAt history (60s cooldown, 5 codes / 15 minutes).
   481→model StaffLoginCode {
   482→  id        String       @id @default(cuid())
   483→  accountId String
   484→  account   StaffAccount @relation(fields: [accountId], references: [id], onDelete: Cascade)
   485→  /// sha256 hex digest of the 6-digit code. Never the plaintext.
   486→  codeHash  String
   487→  /// After this instant the code is dead even if unused.
   488→  expiresAt DateTime
   489→  /// Single-use latch: set by the ONE verification transaction that wins.
   490→  consumedAt DateTime?
   491→  /// Failed verification attempts against this code (max 5).
   492→  attemptCount Int      @default(0)
   493→  createdAt DateTime     @default(now())
   494→
   495→  @@index([accountId, createdAt])
   496→  @@map("staff_login_codes")
   497→}

   499→/// One authenticated Staff Portal session (Stage 7). The browser holds only
   500→/// an opaque random token in an HttpOnly cookie; the server stores ONLY the
   501→/// sha256 hash of that token, so a database read can never mint a session.
   502→/// Sessions expire (7-day MVP default), can be revoked (logout — the DB row
   503→/// is revoked, not merely the cookie dropped) and resolve only for ACTIVE
   504→/// accounts, so a suspended Staff member loses portal access immediately.
   505→model StaffSession {
   506→  id        String       @id @default(cuid())
   507→  accountId String
   508→  account   StaffAccount @relation(fields: [accountId], references: [id], onDelete: Cascade)
   509→  /// sha256 hex digest of the cookie token. Unique — lookup key.
   510→  tokenHash String       @unique
   511→  createdAt DateTime     @default(now())
   512→  /// Hard expiry checked on every resolution (MVP: 7 days, configurable).
   513→  expiresAt DateTime
   514→  /// Logout / future suspension revokes the row — cookie deletion alone is
   515→  /// never trusted.
   516→  revokedAt DateTime?
   517→  /// Refreshed opportunistically by /api/staff/me.
   518→  lastSeenAt DateTime?
   519→
   520→  @@index([accountId])
   521→  @@map("staff_sessions")
   522→}

   545→/// lazily when an eligible (APPROVED_STAFF with a Staff row) staff member
   546→/// first opens the onboarding stage (src/server/services/onboarding.ts).
   547→///
   548→/// Onboarding is deliberately NOT part of the recruitment state machine:
   549→/// it adds no CandidateStatus value, no RecruitmentStage value and no
   550→/// StageEvent — the six-stage recruitment stepper and the append-only
   551→/// StageEvent timeline stay recruitment-only. Auditability comes from this
   552→/// record itself: startedAt, the per-step StaffOnboardingStep completion
   553→/// rows, the final completedAt and the pinned content version.
   554→///
   555→/// Opening onboarding NEVER completes a step and never completes onboarding;
   556→/// step completion and the final acknowledgement are explicit, server-
   557→/// validated actions. `currentStepId` is the resume point (the first
   558→/// not-yet-completed step id from src/config/onboarding.ts), derived on
   559→/// write so "continue where you stopped" survives content reordering.
   560→model StaffOnboarding {
   561→  id          String @id @default(cuid())
   562→  /// One onboarding journey per Staff record.
   563→  staffId     String @unique
   564→  staff       Staff  @relation(fields: [staffId], references: [id], onDelete: Cascade)
   565→  /// Onboarding-content version pinned at entry ("1.0",
   566→  /// src/config/onboarding.ts); re-pinned at completion so the row records
   567→  /// which set of instructions the staff member completed.
   568→  version     String
   569→  /// Resume point: first not-yet-completed step (stable config id, e.g.
   570→  /// "welcome"). Derived on write — never trusted from the client.
   571→  currentStepId String?
   572→  /// When the staff member first opened onboarding (lazy entry).
   573→  startedAt   DateTime
   574→  /// When the staff member explicitly completed onboarding. Null until the
   575→  /// completion action is accepted by the server; the exactly-once latch
   576→  /// for completion is `completedAt = null` → set.
   577→  completedAt DateTime?
   578→  createdAt   DateTime @default(now())
   579→  updatedAt   DateTime @updatedAt
   580→
   581→  steps StaffOnboardingStep[]
   582→
   583→  @@map("staff_onboarding")
   584→}

   586→/// Transactional email delivery record (outbox pattern, brief §9).
   587→///
   588→/// One row per lifecycle notification INTENT. Rows are created by the
   589→/// central notification service in the SAME database transaction as the
   590→/// meaningful state change they belong to (brief §13), so a committed
   591→/// business event always has a corresponding notification intent even if
   592→/// the email provider is down. The unique `dedupeKey` is the exactly-once
   593→/// latch: user double-clicks, request retries, concurrent requests and
   594→/// route refreshes can never queue a duplicate email (brief §11).
   595→///
   596→/// The actual Mailjet network call happens AFTER the business transaction
   597→/// commits — never while the transaction holds database locks (§16). A
   598→/// Mailjet outage therefore marks rows FAILED/PENDING but can never roll
   599→/// back or corrupt recruitment state (§12/§14).
   600→///
   601→/// No credentials are ever stored here; no rendered HTML is stored — the
   602→/// email body is re-derived from immutable snapshot data at send time so
   603→/// retries stay faithful (staff/appointment/agreement/attempt snapshot
   604→/// rows are immutable by schema design).
   605→model NotificationDelivery {
   606→  id        String             @id @default(cuid())
   607→  /// The candidate the notification belongs to, where applicable. Cascade
   608→  /// delete mirrors every other candidate-owned record.
   609→  candidateId String?
   610→  candidate Candidate?         @relation(fields: [candidateId], references: [id], onDelete: Cascade)
   611→  /// Stable lifecycle key (NotificationType).
   612→  type      NotificationType
   613→  /// Stable idempotency key "<TYPE>:<immutable entity id>" (e.g.
   614→  /// "APPLICATION_RECEIVED:<candidateId>", "ASSESSMENT_PASSED:<attemptId>",
   615→  /// "STAFF_ACTIVATED:<staff row id>"). Unique — the exactly-once latch.
   616→  dedupeKey String             @unique
   617→  /// Recipient email snapshotted at enqueue time (never re-read at send).
   618→  recipientEmail String
   619→  /// Code-owned template registry version used to render the email
   620→  /// (src/server/notifications/notification-templates.ts).
   621→  templateVersion String
   622→  /// Email provider handling delivery (MVP: "mailjet").
   623→  provider  String
   624→  /// Provider message id, when the provider accepted a send.
   625→  providerMessageId String?
   626→  /// PENDING until the provider accepts a send; SENT once accepted
   627→  /// (NOT a claim of inbox delivery — see NotificationStatus).
   628→  status    NotificationStatus @default(PENDING)
   629→  /// Number of delivery attempts made (bounded by config).
   630→  attemptCount Int             @default(0)
   631→  /// When the provider accepted the send (status became SENT).
   632→  sentAt    DateTime?
   633→  /// Last delivery attempt, successful or not.
   634→  lastAttemptAt DateTime?
   635→  /// Safe error code from the last failed attempt (HTTP status class or
   636→  /// NETWORK_ERROR/TIMEOUT). Never contains credentials or bodies.
   637→  lastErrorCode String?
   638→  /// Safe error message from the last failed attempt.
   639→  lastErrorMessage String?
   640→  createdAt DateTime           @default(now())
   641→  updatedAt DateTime           @updatedAt
   642→
   643→  @@index([status])
   644→  @@index([candidateId])
   645→  @@map("notification_deliveries")
   646→}

   648→/// Completion record for one onboarding step. Append-only: one row per
   649→/// completed step per onboarding journey (enforced by the unique
   650→/// constraint); rows are never edited or deleted. Steps are sequential —
   651→/// order is enforced server-side against src/config/onboarding.ts, never
   652→/// trusted from the client.
   653→model StaffOnboardingStep {
   654→  id           String          @id @default(cuid())
   655→  onboardingId String
   656→  onboarding   StaffOnboarding @relation(fields: [onboardingId], references: [id], onDelete: Cascade)
   657→  /// Stable step id from src/config/onboarding.ts (e.g. "welcome").
   658→  stepId       String
   659→  completedAt  DateTime        @default(now())
   660→
   661→  @@unique([onboardingId, stepId])
   662→  @@index([onboardingId])
   663→  @@map("staff_onboarding_steps")
   664→}
