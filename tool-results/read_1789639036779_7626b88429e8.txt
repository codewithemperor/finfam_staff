     2→// FinFam Recruitment Portal — schema
     3→//
     4→// Implemented stages: Application (scaffold), FSOW Orientation (stage 1),
     5→// Automated Interview Assessment (stage 2), Responsibility & Reward
     6→// Agreement (stage 3 — persisted as the OFFER stage/statuses; the
     7→// candidate-facing label is "Agreement", see src/config/recruitment.ts),
     8→// the Provisional Appointment (stage 4 — lazy issuance, one letter per
     9→// candidate, see src/server/services/appointment.ts), Staff Activation
    10→// (stage 5 — explicit in-portal activation, one Staff identity per
    11→// candidate, see src/server/services/activation.ts), and Staff Onboarding
    12→// (stage 6 — STAFF-SIDE operational briefing after recruitment completes:
    13→// StaffOnboarding + StaffOnboardingStep, see src/server/services/
    14→// onboarding.ts). Onboarding deliberately adds NO CandidateStatus value
    15→// and NO RecruitmentStage value — APPROVED_STAFF remains the terminal
    16→// recruitment status, and StageEvent stays recruitment-only.
    17→//
    18→// Staff Platform (post-recruitment, ongoing): Stage 7 added the
    19→// authenticated Staff Portal (StaffAccount/StaffLoginCode/StaffSession —
    20→// email-OTP auth, DB-backed sessions), Stage 8 adds Daily Reporting &
    21→// Accountability (DailyReport — one self-reported activity report per
    22→// Staff member per Africa/Lagos reporting date, see
    23→// src/server/services/daily-reports.ts) and Stage 9 adds Staff Training
    24→// (StaffTrainingEnrollment/StaffTrainingLessonProgress — source-controlled
    25→// course content with persisted, version-pinned Staff progress, see
    26→// src/server/services/staff-training.ts). None of these modules touch the
    27→// recruitment state machine.
    28→//
    29→// Cross-cutting platform service: NotificationDelivery (brief
    30→// "notification_service") records transactional email intents for
    31→// lifecycle events via the central notification service
    32→// (src/server/notifications/) — outbox pattern, dedupe-key exactly-once,
    33→// Mailjet delivery AFTER the business transaction commits.
    34→// See docs/CONTENT-QUESTIONS.md before extending: stage rules are
    35→// finalized stage-by-stage in the product planning phase.

    90→/// Delivery status of a transactional notification.

   104→/// state change — there is deliberately no per-page-view or per-step key.

   106→/// Account provisioning (activation transaction + backfill, Stage 7):

   121→  /// Staff Portal email-OTP login code (Stage 7). One-shot semantics: the
   122→  /// plaintext code exists only in memory between creation and the single
   123→  /// delivery attempt, so the payload can never be re-derived for a retry —
   124→  /// a failed delivery is terminal for that code and the user requests a
   125→  /// fresh one. Codes themselves are stored hashed on staff_login_codes.
   126→  STAFF_LOGIN_CODE
   127→
   128→  /// Daily-report receipt (Stage 8). Queued inside the ONE submission
   129→  /// transaction that wins the DRAFT → SUBMITTED latch, keyed
   130→  /// "DAILY_REPORT_SUBMITTED:<reportId>" — exactly once per submitted
   131→  /// report. A concise transactional receipt (report date + received
   132→  /// confirmation), NOT a metrics digest and never a reminder — no
   133→  /// DAILY_REPORT_REMINDER exists until a deadline policy is confirmed.
   134→  DAILY_REPORT_SUBMITTED
   135→
   136→  /// Admin console email-OTP sign-in code (Stage 11). One-shot semantics,
   137→  /// identical in kind to STAFF_LOGIN_CODE: the plaintext code exists only
   138→  /// in memory between creation and the single delivery attempt, so the
   139→  /// payload can never be re-derived for a retry — a failed delivery is
   140→  /// terminal for that code. Codes are stored hashed on admin_login_codes.
   141→  /// The recipient is the AdminAccount email (no Candidate row exists for
   142→  /// administrators — notification_deliveries.candidateId stays null, the
   143→  /// recipient identity accurately represents the real recipient).
   144→  ADMIN_LOGIN_CODE

   166→/// One Staff member's reported activity for ONE reporting date (Stage 8 —
   167→/// Daily Reporting & Accountability, an ongoing STAFF PLATFORM module that
   168→/// adds NO recruitment state: no CandidateStatus, no RecruitmentStage, no
   169→/// StageEvent). Flat typed columns carry the nine approved daily metrics
   170→/// (the four categories taught by the onboarding "accountability" step) —
   171→/// not a JSON blob and not metric rows — because the set is small, fixed
   172→/// and individually validated/displayed. `@@unique([staffId, reportDate])`
   173→/// is the one-report-per-staff-per-date rule and the concurrency backstop:
   174→/// create-or-get races lose the P2002 and re-read the winner.
   175→///
   176→/// Facts kept separate on purpose (brief §4): `reportDate` is the calendar
   177→/// date the reported work occurred (Africa/Lagos — stored as that calendar
   178→/// DATE, see src/lib/dates/lagos.ts), while `submittedAt` is the server
   179→/// instant the report was finalised; createdAt/updatedAt are row
   180→/// bookkeeping and never stand in for the activity date.
   181→///
   182→/// The data is STAFF SELF-REPORTED (brief §22): it is accountability/
   183→/// activity data only — never presented as independently verified, never
   184→/// used for compensation (no ₦650 payroll path exists anywhere), and it
   185→/// contains NO person-level member/prospect records (that is a future
   186→/// module). `version` pins the report-schema interpretation at creation.
   187→model DailyReport {
   188→  id      String            @id @default(cuid())
   189→  /// Owning Staff row — resolved from the authenticated session chain,
   190→  /// NEVER from client input (brief §31/§32).
   191→  staffId String
   192→  staff   Staff             @relation(fields: [staffId], references: [id], onDelete: Cascade)
   193→  /// Calendar date the reported work occurred (Africa/Lagos). Stored as
   194→  /// that DATE (UTC midnight of the Lagos date, src/lib/dates/lagos.ts).
   195→  reportDate DateTime        @db.Date
   196→  /// Report-schema version pinned at creation ("1.0",
   197→  /// src/config/daily-report.ts). Immutable.
   198→  version String
   199→  status  DailyReportStatus @default(DRAFT)
   200→
   201→  // ── Prospects ─────────────────────────────────────────────────────────
   202→  individualsContacted Int @default(0)
   203→  groupsApproached     Int @default(0)
   204→  followUpActivities   Int @default(0)
   205→  // ── Member Growth ─────────────────────────────────────────────────────
   206→  newRegistrations          Int @default(0)
   207→  establishedMembersCreated Int @default(0)
   208→  // ── Savings Activities ────────────────────────────────────────────────
   209→  firstDepositsAchieved Int @default(0)
   210→  savingsGoalsActivated Int @default(0)
   211→  // ── Community Activities ──────────────────────────────────────────────
   212→  memberEngagement Int @default(0)
   213→  connectionsBuilt Int @default(0)
   214→
   215→  /// Optional plain-text context the counts cannot capture. Trimmed,
   216→  /// max 1000 chars, server-validated. Plain text only — never PII.
   217→  activitySummary String?
   218→
   219→  /// Server finalisation instant. Null while DRAFT; the one-way latch
   220→  /// (status DRAFT → SUBMITTED via a conditional update) stamps it exactly
   221→  /// once, so repeated/concurrent submits yield ONE timestamp.
   222→  submittedAt DateTime?
   223→  createdAt   DateTime  @default(now())
   224→  updatedAt   DateTime  @updatedAt
   225→
   226→  /// One Staff member + one activity date = one report.
   227→  @@unique([staffId, reportDate])
   228→  /// Stage 11 admin console: month/day filtering across ALL staff on
   229→  /// /admin/reports cannot use the staff-prefixed unique above.
   230→  @@index([reportDate])
   231→  @@map("daily_reports")
   232→}

   425→/// The candidate's issued Provisional Letter of Appointment — stage 4 of
   426→/// the journey. Issued lazily and exactly once, server-side, when an
   427→/// OFFER_ACCEPTED candidate first opens the appointment stage
   428→/// (src/server/services/appointment.ts); viewing again is a read-only
   429→/// record and NEVER re-issues or mutates the letter.
   430→///
   431→/// Hybrid historical-integrity model: the columns below are the immutable,
   432→/// candidate-specific SNAPSHOT (who, which letter, when, under which
   433→/// content version) while the letter TEXT itself lives in an append-only
   434→/// content registry (src/config/appointment.ts, APPOINTMENT_CONTENT_
   435→/// VERSIONS) resolved by `version`. An appointment issued under v1.0 must
   436→/// stay reproducible as v1.0 forever — later wording changes append new
   437→/// registry versions and never touch issued ones. There is deliberately no
   438→/// separate effectiveDate column: the letter date IS `issuedAt` for the
   439→/// MVP (docs/CONTENT-QUESTIONS.md §23 keeps the distinction open).
   440→model Appointment {
   441→  id          String    @id @default(cuid())
   442→  candidateId String    @unique
   443→  candidate   Candidate @relation(fields: [candidateId], references: [id], onDelete: Cascade)
   444→  /// Official letter reference in the FFL/HR/PLA/NNN format. Allocated
   445→  /// server-side from the AppointmentSequence inside the issuance
   446→  /// transaction — never derived from client input, never from count(*).
   447→  /// The unique constraint is the final backstop against duplicates.
   448→  appointmentReference String @unique
   449→  /// Pinned content-registry version (e.g. "1.0") the letter was issued
   450→  /// under. Immutable after issuance.
   451→  version     String
   452→  /// Printed candidate name, snapshotted from Candidate.fullName at
   453→  /// issuance. Never read from the client at render time.
   454→  printedName String
   455→  /// Official appointment designation, snapshotted at issuance. The
   456→  /// approved MVP value is "Remote Staff" (the designation used by the
   457→  /// official Provisional Letter of Appointment) — not renamed to any
   458→  /// other role title (docs/CONTENT-QUESTIONS.md §21).
   459→  roleTitle   String
   460→  /// Server issuance timestamp. Doubles as the printed letter date for
   461→  /// the MVP; no separate effectiveDate exists (§23).
   462→  issuedAt    DateTime
   463→  createdAt   DateTime  @default(now())
   464→  updatedAt   DateTime  @updatedAt
   465→
   466→  @@map("appointments")
   467→}

   469→/// Server-controlled sequence behind the FFL/HR/PLA/NNN appointment
   470→/// references. A single row per sequence: `value` holds the LAST allocated
   471→/// number; allocation is an atomic `UPDATE ... SET value = value + 1
   472→/// RETURNING value` inside the issuance transaction, so concurrent
   473→/// issuances can never receive the same number (the unique constraint on
   474→/// appointments.appointmentReference stays the final backstop).
   475→///
   476→/// Initialisation: the supplied official sample letter carries
   477→/// FFL/HR/PLA/012 and is treated as the most recently existing appointment
   478→/// (an externally issued document — it is NOT written into the database).
   479→/// The sequence therefore starts at 12, making the first system-issued
   480→/// reference FFL/HR/PLA/013. Documented assumption; see
   481→/// docs/CONTENT-QUESTIONS.md §22.
   482→model AppointmentSequence {
   483→  id    String @id @default(cuid())
   484→  /// Singleton sequence name (one row: "PROVISIONAL_APPOINTMENT").
   485→  name  String @unique
   486→  /// Last allocated sequence number (initialised to 12 by migration).
   487→  value Int
   488→
   489→  @@map("appointment_sequences")
   490→}

   702→/// Transactional email delivery record (outbox pattern, brief §9).
