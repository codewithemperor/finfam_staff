   156→   156→/// One Staff member's reported activity for ONE reporting date (Stage 8 —
   157→   157→/// Daily Reporting & Accountability, an ongoing STAFF PLATFORM module that
   158→   158→/// adds NO recruitment state: no CandidateStatus, no RecruitmentStage, no
   159→   159→/// StageEvent). Flat typed columns carry the nine approved daily metrics
   160→   160→/// (the four categories taught by the onboarding "accountability" step) —
   161→   161→/// not a JSON blob and not metric rows — because the set is small, fixed
   162→   162→/// and individually validated/displayed. `@@unique([staffId, reportDate])`
   163→   163→/// is the one-report-per-staff-per-date rule and the concurrency backstop:
   164→   164→/// create-or-get races lose the P2002 and re-read the winner.
   165→   165→///
   166→   166→/// Facts kept separate on purpose (brief §4): `reportDate` is the calendar
   167→   167→/// date the reported work occurred (Africa/Lagos — stored as that calendar
   168→   168→/// DATE, see src/lib/dates/lagos.ts), while `submittedAt` is the server
   169→   169→/// instant the report was finalised; createdAt/updatedAt are row
   170→   170→/// bookkeeping and never stand in for the activity date.
   171→   171→///
   172→   172→/// The data is STAFF SELF-REPORTED (brief §22): it is accountability/
   173→   173→/// activity data only — never presented as independently verified, never
   174→   174→/// used for compensation (no ₦650 payroll path exists anywhere), and it
   175→   175→/// contains NO person-level member/prospect records (that is a future
   176→   176→/// module). `version` pins the report-schema interpretation at creation.
   177→   177→model DailyReport {
   178→   178→  id      String            @id @default(cuid())
   179→   179→  /// Owning Staff row — resolved from the authenticated session chain,
   180→   180→  /// NEVER from client input (brief §31/§32).
   181→   181→  staffId String
   182→   182→  staff   Staff             @relation(fields: [staffId], references: [id], onDelete: Cascade)
   183→   183→  /// Calendar date the reported work occurred (Africa/Lagos). Stored as
   184→   184→  /// that DATE (UTC midnight of the Lagos date, src/lib/dates/lagos.ts).
   185→   185→  reportDate DateTime        @db.Date
   186→   186→  /// Report-schema version pinned at creation ("1.0",
   187→   187→  /// src/config/daily-report.ts). Immutable.
   188→   188→  version String
   189→   189→  status  DailyReportStatus @default(DRAFT)
   190→   190→
   191→   191→  // ── Prospects ─────────────────────────────────────────────────────────
   192→   192→  individualsContacted Int @default(0)
   193→   193→  groupsApproached     Int @default(0)
   194→   194→  followUpActivities   Int @default(0)
   195→   195→  // ── Member Growth ─────────────────────────────────────────────────────
   196→   196→  newRegistrations          Int @default(0)
   197→   197→  establishedMembersCreated Int @default(0)
   198→   198→  // ── Savings Activities ────────────────────────────────────────────────
   199→   199→  firstDepositsAchieved Int @default(0)
   200→   200→  savingsGoalsActivated Int @default(0)
   201→   201→  // ── Community Activities ──────────────────────────────────────────────
   202→   202→  memberEngagement Int @default(0)
   203→   203→  connectionsBuilt Int @default(0)
   204→   204→
   205→   205→  /// Optional plain-text context the counts cannot capture. Trimmed,
   206→   206→  /// max 1000 chars, server-validated. Plain text only — never PII.
   207→   207→  activitySummary String?
   208→   208→
   209→   209→  /// Server finalisation instant. Null while DRAFT; the one-way latch
   210→   210→  /// (status DRAFT → SUBMITTED via a conditional update) stamps it exactly
   211→   211→  /// once, so repeated/concurrent submits yield ONE timestamp.
   212→   212→  submittedAt DateTime?
   213→   213→  createdAt   DateTime  @default(now())
   214→   214→  updatedAt   DateTime  @updatedAt
   215→   215→
   216→   216→  /// One Staff member + one activity date = one report.
   217→   217→  @@unique([staffId, reportDate])
   218→   218→  @@map("daily_reports")
   219→   219→}

   412→   412→/// The candidate's issued Provisional Letter of Appointment — stage 4 of
   413→   413→/// the journey. Issued lazily and exactly once, server-side, when an
   414→   414→/// OFFER_ACCEPTED candidate first opens the appointment stage
   415→   415→/// (src/server/services/appointment.ts); viewing again is a read-only
   416→   416→/// record and NEVER re-issues or mutates the letter.
   417→   417→///
   418→   418→/// Hybrid historical-integrity model: the columns below are the immutable,
   419→   419→/// candidate-specific SNAPSHOT (who, which letter, when, under which
   420→   420→/// content version) while the letter TEXT itself lives in an append-only
   421→   421→/// content registry (src/config/appointment.ts, APPOINTMENT_CONTENT_
   422→   422→/// VERSIONS) resolved by `version`. An appointment issued under v1.0 must
   423→   423→/// stay reproducible as v1.0 forever — later wording changes append new
   424→   424→/// registry versions and never touch issued ones. There is deliberately no
   425→   425→/// separate effectiveDate column: the letter date IS `issuedAt` for the
   426→   426→/// MVP (docs/CONTENT-QUESTIONS.md §23 keeps the distinction open).
   427→   427→model Appointment {
   428→   428→  id          String    @id @default(cuid())
   429→   429→  candidateId String    @unique
   430→   430→  candidate   Candidate @relation(fields: [candidateId], references: [id], onDelete: Cascade)
   431→   431→  /// Official letter reference in the FFL/HR/PLA/NNN format. Allocated
   432→   432→  /// server-side from the AppointmentSequence inside the issuance
   433→   433→  /// transaction — never derived from client input, never from count(*).
   434→   434→  /// The unique constraint is the final backstop against duplicates.
   435→   435→  appointmentReference String @unique
   436→   436→  /// Pinned content-registry version (e.g. "1.0") the letter was issued
   437→   437→  /// under. Immutable after issuance.
   438→   438→  version     String
   439→   439→  /// Printed candidate name, snapshotted from Candidate.fullName at
   440→   440→  /// issuance. Never read from the client at render time.
   441→   441→  printedName String
   442→   442→  /// Official appointment designation, snapshotted at issuance. The
   443→   443→  /// approved MVP value is "Remote Staff" (the designation used by the
   444→   444→  /// official Provisional Letter of Appointment) — not renamed to any
   445→   445→  /// other role title (docs/CONTENT-QUESTIONS.md §21).
   446→   446→  roleTitle   String
   447→   447→  /// Server issuance timestamp. Doubles as the printed letter date for
   448→   448→  /// the MVP; no separate effectiveDate exists (§23).
