     4→// Implemented stages: Application (scaffold), FSOW Orientation (stage 1),
     5→// Automated Interview Assessment (stage 2), Responsibility & Reward

     8→// the Provisional Appointment (stage 4 — lazy issuance, one letter per
     9→// candidate, see src/server/services/appointment.ts), Staff Activation

    11→// candidate, see src/server/services/activation.ts), and Staff Onboarding
    12→// (stage 6 — STAFF-SIDE operational briefing after recruitment completes:
    13→// StaffOnboarding + StaffOnboardingStep, see src/server/services/
    14→// onboarding.ts). Onboarding deliberately adds NO CandidateStatus value
    15→// and NO RecruitmentStage value — APPROVED_STAFF remains the terminal
    16→// recruitment status, and StageEvent stays recruitment-only.
    17→//
    18→// Staff Platform (post-recruitment, ongoing): Stage 7 added the
    19→// authenticated Staff Portal (StaffAccount/StaffLoginCode/StaffSession —
    20→// email-OTP auth, DB-backed sessions) and Stage 8 adds Daily Reporting &
    21→// Accountability (DailyReport — one self-reported activity report per
    22→// Staff member per Africa/Lagos reporting date, see
    23→// src/server/services/daily-reports.ts). Neither module touches the
    24→// recruitment state machine.
    25→//
    26→// Cross-cutting platform service: NotificationDelivery (brief
    27→// "notification_service") records transactional email intents for
    28→// lifecycle events via the central notification service
    29→// (src/server/notifications/) — outbox pattern, dedupe-key exactly-once,
    30→// Mailjet delivery AFTER the business transaction commits.

    87→/// Delivery status of a transactional notification.
    88→///
    89→/// SENT means the email provider ACCEPTED the transactional send request —

   143→/// Lifecycle of one daily activity report (Stage 8). Deliberately the

   145→/// SUBMITTED (final, read-only permanently in the Staff MVP). There is

   150→  SUBMITTED

   153→/// One Staff member's reported activity for ONE reporting date (Stage 8 —
   154→/// Daily Reporting & Accountability, an ongoing STAFF PLATFORM module that
   155→/// adds NO recruitment state: no CandidateStatus, no RecruitmentStage, no
   156→/// StageEvent). Flat typed columns carry the nine approved daily metrics
   157→/// (the four categories taught by the onboarding "accountability" step) —
   158→/// not a JSON blob and not metric rows — because the set is small, fixed
   159→/// and individually validated/displayed. `@@unique([staffId, reportDate])`
   160→/// is the one-report-per-staff-per-date rule and the concurrency backstop:
   161→/// create-or-get races lose the P2002 and re-read the winner.
   162→///
   163→/// Facts kept separate on purpose (brief §4): `reportDate` is the calendar
   164→/// date the reported work occurred (Africa/Lagos — stored as that calendar
   165→/// DATE, see src/lib/dates/lagos.ts), while `submittedAt` is the server
   166→/// instant the report was finalised; createdAt/updatedAt are row
   167→/// bookkeeping and never stand in for the activity date.
   168→///
   169→/// The data is STAFF SELF-REPORTED (brief §22): it is accountability/
   170→/// activity data only — never presented as independently verified, never
   171→/// used for compensation (no ₦650 payroll path exists anywhere), and it
   172→/// contains NO person-level member/prospect records (that is a future
   173→/// module). `version` pins the report-schema interpretation at creation.
   174→model DailyReport {
   175→  id      String            @id @default(cuid())
   176→  /// Owning Staff row — resolved from the authenticated session chain,
   177→  /// NEVER from client input (brief §31/§32).
   178→  staffId String
   179→  staff   Staff             @relation(fields: [staffId], references: [id], onDelete: Cascade)
   180→  /// Calendar date the reported work occurred (Africa/Lagos). Stored as
   181→  /// that DATE (UTC midnight of the Lagos date, src/lib/dates/lagos.ts).
   182→  reportDate DateTime        @db.Date
   183→  /// Report-schema version pinned at creation ("1.0",
   184→  /// src/config/daily-report.ts). Immutable.
   185→  version String
   186→  status  DailyReportStatus @default(DRAFT)
   187→
   188→  // ── Prospects ─────────────────────────────────────────────────────────
   189→  individualsContacted Int @default(0)
   190→  groupsApproached     Int @default(0)
   191→  followUpActivities   Int @default(0)
   192→  // ── Member Growth ─────────────────────────────────────────────────────
   193→  newRegistrations          Int @default(0)
   194→  establishedMembersCreated Int @default(0)
   195→  // ── Savings Activities ────────────────────────────────────────────────
   196→  firstDepositsAchieved Int @default(0)
   197→  savingsGoalsActivated Int @default(0)
   198→  // ── Community Activities ──────────────────────────────────────────────
   199→  memberEngagement Int @default(0)
   200→  connectionsBuilt Int @default(0)
   201→
   202→  /// Optional plain-text context the counts cannot capture. Trimmed,
   203→  /// max 1000 chars, server-validated. Plain text only — never PII.
   204→  activitySummary String?
   205→
   206→  /// Server finalisation instant. Null while DRAFT; the one-way latch
   207→  /// (status DRAFT → SUBMITTED via a conditional update) stamps it exactly
   208→  /// once, so repeated/concurrent submits yield ONE timestamp.
   209→  submittedAt DateTime?
   210→  createdAt   DateTime  @default(now())
   211→  updatedAt   DateTime  @updatedAt
   212→
   213→  /// One Staff member + one activity date = one report.
   214→  @@unique([staffId, reportDate])
   215→  @@map("daily_reports")
   216→}

   409→/// The candidate's issued Provisional Letter of Appointment — stage 4 of
   410→/// the journey. Issued lazily and exactly once, server-side, when an
   411→/// OFFER_ACCEPTED candidate first opens the appointment stage
   412→/// (src/server/services/appointment.ts); viewing again is a read-only
   413→/// record and NEVER re-issues or mutates the letter.
   414→///
   415→/// Hybrid historical-integrity model: the columns below are the immutable,
   416→/// candidate-specific SNAPSHOT (who, which letter, when, under which
   417→/// content version) while the letter TEXT itself lives in an append-only
   418→/// content registry (src/config/appointment.ts, APPOINTMENT_CONTENT_
   419→/// VERSIONS) resolved by `version`. An appointment issued under v1.0 must
   420→/// stay reproducible as v1.0 forever — later wording changes append new
   421→/// registry versions and never touch issued ones. There is deliberately no
   422→/// separate effectiveDate column: the letter date IS `issuedAt` for the
   423→/// MVP (docs/CONTENT-QUESTIONS.md §23 keeps the distinction open).
   424→model Appointment {
   425→  id          String    @id @default(cuid())
   426→  candidateId String    @unique
   427→  candidate   Candidate @relation(fields: [candidateId], references: [id], onDelete: Cascade)
   428→  /// Official letter reference in the FFL/HR/PLA/NNN format. Allocated
   429→  /// server-side from the AppointmentSequence inside the issuance
   430→  /// transaction — never derived from client input, never from count(*).
   431→  /// The unique constraint is the final backstop against duplicates.
   432→  appointmentReference String @unique
   433→  /// Pinned content-registry version (e.g. "1.0") the letter was issued
   434→  /// under. Immutable after issuance.
   435→  version     String
   436→  /// Printed candidate name, snapshotted from Candidate.fullName at
   437→  /// issuance. Never read from the client at render time.
   438→  printedName String
   439→  /// Official appointment designation, snapshotted at issuance. The
   440→  /// approved MVP value is "Remote Staff" (the designation used by the
   441→  /// official Provisional Letter of Appointment) — not renamed to any
   442→  /// other role title (docs/CONTENT-QUESTIONS.md §21).
   443→  roleTitle   String
   444→  /// Server issuance timestamp. Doubles as the printed letter date for
   445→  /// the MVP; no separate effectiveDate exists (§23).
   446→  issuedAt    DateTime
   447→  createdAt   DateTime  @default(now())
   448→  updatedAt   DateTime  @updatedAt
   449→
   450→  @@map("appointments")
   451→}

   453→/// Server-controlled sequence behind the FFL/HR/PLA/NNN appointment
   454→/// references. A single row per sequence: `value` holds the LAST allocated
   455→/// number; allocation is an atomic `UPDATE ... SET value = value + 1
   456→/// RETURNING value` inside the issuance transaction, so concurrent
   457→/// issuances can never receive the same number (the unique constraint on
   458→/// appointments.appointmentReference stays the final backstop).
   459→///
   460→/// Initialisation: the supplied official sample letter carries
   461→/// FFL/HR/PLA/012 and is treated as the most recently existing appointment
   462→/// (an externally issued document — it is NOT written into the database).
   463→/// The sequence therefore starts at 12, making the first system-issued
   464→/// reference FFL/HR/PLA/013. Documented assumption; see
   465→/// docs/CONTENT-QUESTIONS.md §22.
   466→model AppointmentSequence {
   467→  id    String @id @default(cuid())
   468→  /// Singleton sequence name (one row: "PROVISIONAL_APPOINTMENT").
   469→  name  String @unique
   470→  /// Last allocated sequence number (initialised to 12 by migration).
   471→  value Int
   472→
   473→  @@map("appointment_sequences")
   474→}

   534→/// per Staff record, created in the SAME transaction as the Staff row by
   535→/// staff activation (and idempotently by the backfill script for pre-existing
   536→/// Staff). The account owns login codes and portal sessions and carries the

   565→/// One emailed email-OTP login code for a Staff Portal account (Stage 7).
   566→///
   567→/// Security properties (brief §6/§7/§10): the 6-digit code is generated with
   568→/// cryptographically secure randomness and stored ONLY as a sha256 hash —
   569→/// the plaintext lives in memory just long enough to render the one-shot
   570→/// notification payload and is never persisted or logged. A code is
   571→/// single-use (consumedAt latch), expires after 10 minutes, tolerates at

   681→/// Transactional email delivery record (outbox pattern, brief §9).
   682→///
   683→/// One row per lifecycle notification INTENT. Rows are created by the
   684→/// central notification service in the SAME database transaction as the
   685→/// meaningful state change they belong to (brief §13), so a committed
   686→/// business event always has a corresponding notification intent even if
   687→/// the email provider is down. The unique `dedupeKey` is the exactly-once
   688→/// latch: user double-clicks, request retries, concurrent requests and
   689→/// route refreshes can never queue a duplicate email (brief §11).
   690→///
   691→/// The actual Mailjet network call happens AFTER the business transaction
   692→/// commits — never while the transaction holds database locks (§16). A
   693→/// Mailjet outage therefore marks rows FAILED/PENDING but can never roll
   694→/// back or corrupt recruitment state (§12/§14).
   695→///
   696→/// No credentials are ever stored here; no rendered HTML is stored — the
   697→/// email body is re-derived from immutable snapshot data at send time so
   698→/// retries stay faithful (staff/appointment/agreement/attempt snapshot
   699→/// rows are immutable by schema design).
   700→model NotificationDelivery {
   701→  id        String             @id @default(cuid())
   702→  /// The candidate the notification belongs to, where applicable. Cascade
   703→  /// delete mirrors every other candidate-owned record.
   704→  candidateId String?
   705→  candidate Candidate?         @relation(fields: [candidateId], references: [id], onDelete: Cascade)
   706→  /// Stable lifecycle key (NotificationType).
   707→  type      NotificationType
   708→  /// Stable idempotency key "<TYPE>:<immutable entity id>" (e.g.
   709→  /// "APPLICATION_RECEIVED:<candidateId>", "ASSESSMENT_PASSED:<attemptId>",
   710→  /// "STAFF_ACTIVATED:<staff row id>"). Unique — the exactly-once latch.
   711→  dedupeKey String             @unique
   712→  /// Recipient email snapshotted at enqueue time (never re-read at send).
   713→  recipientEmail String
   714→  /// Code-owned template registry version used to render the email
   715→  /// (src/server/notifications/notification-templates.ts).
   716→  templateVersion String
   717→  /// Email provider handling delivery (MVP: "mailjet").
   718→  provider  String
   719→  /// Provider message id, when the provider accepted a send.
   720→  providerMessageId String?
   721→  /// PENDING until the provider accepts a send; SENT once accepted
   722→  /// (NOT a claim of inbox delivery — see NotificationStatus).
   723→  status    NotificationStatus @default(PENDING)
   724→  /// Number of delivery attempts made (bounded by config).
   725→  attemptCount Int             @default(0)
   726→  /// When the provider accepted the send (status became SENT).
   727→  sentAt    DateTime?
   728→  /// Last delivery attempt, successful or not.
   729→  lastAttemptAt DateTime?
   730→  /// Safe error code from the last failed attempt (HTTP status class or
   731→  /// NETWORK_ERROR/TIMEOUT). Never contains credentials or bodies.
   732→  lastErrorCode String?
   733→  /// Safe error message from the last failed attempt.
   734→  lastErrorMessage String?
   735→  createdAt DateTime           @default(now())
   736→  updatedAt DateTime           @updatedAt
   737→
   738→  @@index([status])
   739→  @@index([candidateId])
   740→  @@map("notification_deliveries")
   741→}
