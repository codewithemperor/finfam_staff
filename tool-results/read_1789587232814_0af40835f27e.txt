     2→// FinFam Recruitment Portal — schema
     3→//
     4→// Implemented stages: Application (scaffold), FSOW Orientation (stage 1),
     5→// Automated Interview Assessment (stage 2), Responsibility & Reward
     6→// Agreement (stage 3 — persisted as the OFFER stage/statuses; the
     7→// candidate-facing label is "Agreement", see src/config/recruitment.ts),
     8→// the Provisional Appointment (stage 4 — lazy issuance, one letter per
     9→// candidate, see src/server/services/appointment.ts), Staff Activation
    10→// (stage 5 — explicit in-portal activation, one Staff identity per
    11→// candidate, see src/server/services/activation.ts), and Staff Onboarding
    12→// (stage 6 — STAFF-SIDE operational briefing after recruitment completes:
    13→// StaffOnboarding + StaffOnboardingStep, see src/server/services/
    14→// onboarding.ts). Onboarding deliberately adds NO CandidateStatus value
    15→// and NO RecruitmentStage value — APPROVED_STAFF remains the terminal
    16→// recruitment status, and StageEvent stays recruitment-only.
    17→//
    18→// Staff Platform (post-recruitment, ongoing): Stage 7 added the
    19→// authenticated Staff Portal (StaffAccount/StaffLoginCode/StaffSession —
    20→// email-OTP auth, DB-backed sessions), Stage 8 adds Daily Reporting &
    21→// Accountability (DailyReport — one self-reported activity report per
    22→// Staff member per Africa/Lagos reporting date, see
    23→// src/server/services/daily-reports.ts) and Stage 9 adds Staff Training
    24→// (StaffTrainingEnrollment/StaffTrainingLessonProgress — source-controlled
    25→// course content with persisted, version-pinned Staff progress, see
    26→// src/server/services/staff-training.ts). None of these modules touch the
    27→// recruitment state machine.
    28→//
    29→// Cross-cutting platform service: NotificationDelivery (brief
    30→// "notification_service") records transactional email intents for
    31→// lifecycle events via the central notification service
    32→// (src/server/notifications/) — outbox pattern, dedupe-key exactly-once,
    33→// Mailjet delivery AFTER the business transaction commits.
    34→// See docs/CONTENT-QUESTIONS.md before extending: stage rules are
    35→// finalized stage-by-stage in the product planning phase.

    90→/// Delivery status of a transactional notification.
    91→///
    92→/// SENT means the email provider ACCEPTED the transactional send request —
    93→/// it is deliberately NOT a claim of inbox delivery. Actual delivery, bounces
    94→/// and spam events require provider webhooks, which are out of MVP scope

    96→/// rejected or errored; PENDING means not yet (successfully) sent.

   146→/// Lifecycle of one daily activity report (Stage 8). Deliberately the

   153→  SUBMITTED

   156→/// One Staff member's reported activity for ONE reporting date (Stage 8 —
   157→/// Daily Reporting & Accountability, an ongoing STAFF PLATFORM module that
   158→/// adds NO recruitment state: no CandidateStatus, no RecruitmentStage, no
   159→/// StageEvent). Flat typed columns carry the nine approved daily metrics
   160→/// (the four categories taught by the onboarding "accountability" step) —
   161→/// not a JSON blob and not metric rows — because the set is small, fixed
   162→/// and individually validated/displayed. `@@unique([staffId, reportDate])`
   163→/// is the one-report-per-staff-per-date rule and the concurrency backstop:
   164→/// create-or-get races lose the P2002 and re-read the winner.
   165→///
   166→/// Facts kept separate on purpose (brief §4): `reportDate` is the calendar
   167→/// date the reported work occurred (Africa/Lagos — stored as that calendar
   168→/// DATE, see src/lib/dates/lagos.ts), while `submittedAt` is the server
   169→/// instant the report was finalised; createdAt/updatedAt are row
   170→/// bookkeeping and never stand in for the activity date.
   171→///
   172→/// The data is STAFF SELF-REPORTED (brief §22): it is accountability/
   173→/// activity data only — never presented as independently verified, never
   174→/// used for compensation (no ₦650 payroll path exists anywhere), and it
   175→/// contains NO person-level member/prospect records (that is a future
   176→/// module). `version` pins the report-schema interpretation at creation.
   177→model DailyReport {
   178→  id      String            @id @default(cuid())
   179→  /// Owning Staff row — resolved from the authenticated session chain,
   180→  /// NEVER from client input (brief §31/§32).
   181→  staffId String
   182→  staff   Staff             @relation(fields: [staffId], references: [id], onDelete: Cascade)
   183→  /// Calendar date the reported work occurred (Africa/Lagos). Stored as
   184→  /// that DATE (UTC midnight of the Lagos date, src/lib/dates/lagos.ts).
   185→  reportDate DateTime        @db.Date
   186→  /// Report-schema version pinned at creation ("1.0",
   187→  /// src/config/daily-report.ts). Immutable.
   188→  version String
   189→  status  DailyReportStatus @default(DRAFT)
   190→
   191→  // ── Prospects ─────────────────────────────────────────────────────────
   192→  individualsContacted Int @default(0)
   193→  groupsApproached     Int @default(0)
   194→  followUpActivities   Int @default(0)
   195→  // ── Member Growth ─────────────────────────────────────────────────────
   196→  newRegistrations          Int @default(0)
   197→  establishedMembersCreated Int @default(0)
   198→  // ── Savings Activities ────────────────────────────────────────────────
   199→  firstDepositsAchieved Int @default(0)
   200→  savingsGoalsActivated Int @default(0)
   201→  // ── Community Activities ──────────────────────────────────────────────
   202→  memberEngagement Int @default(0)
   203→  connectionsBuilt Int @default(0)
   204→
   205→  /// Optional plain-text context the counts cannot capture. Trimmed,
   206→  /// max 1000 chars, server-validated. Plain text only — never PII.
   207→  activitySummary String?
   208→
   209→  /// Server finalisation instant. Null while DRAFT; the one-way latch
   210→  /// (status DRAFT → SUBMITTED via a conditional update) stamps it exactly
   211→  /// once, so repeated/concurrent submits yield ONE timestamp.
   212→  submittedAt DateTime?
   213→  createdAt   DateTime  @default(now())
   214→  updatedAt   DateTime  @updatedAt
   215→
   216→  /// One Staff member + one activity date = one report.
   217→  @@unique([staffId, reportDate])
   218→  @@map("daily_reports")
   219→}

   412→/// The candidate's issued Provisional Letter of Appointment — stage 4 of
   413→/// the journey. Issued lazily and exactly once, server-side, when an
   414→/// OFFER_ACCEPTED candidate first opens the appointment stage
   415→/// (src/server/services/appointment.ts); viewing again is a read-only
   416→/// record and NEVER re-issues or mutates the letter.
   417→///
   418→/// Hybrid historical-integrity model: the columns below are the immutable,
   419→/// candidate-specific SNAPSHOT (who, which letter, when, under which
   420→/// content version) while the letter TEXT itself lives in an append-only
   421→/// content registry (src/config/appointment.ts, APPOINTMENT_CONTENT_
   422→/// VERSIONS) resolved by `version`. An appointment issued under v1.0 must
   423→/// stay reproducible as v1.0 forever — later wording changes append new
   424→/// registry versions and never touch issued ones. There is deliberately no
   425→/// separate effectiveDate column: the letter date IS `issuedAt` for the
   426→/// MVP (docs/CONTENT-QUESTIONS.md §23 keeps the distinction open).
   427→model Appointment {
   428→  id          String    @id @default(cuid())
   429→  candidateId String    @unique
   430→  candidate   Candidate @relation(fields: [candidateId], references: [id], onDelete: Cascade)
   431→  /// Official letter reference in the FFL/HR/PLA/NNN format. Allocated
   432→  /// server-side from the AppointmentSequence inside the issuance
   433→  /// transaction — never derived from client input, never from count(*).
   434→  /// The unique constraint is the final backstop against duplicates.
   435→  appointmentReference String @unique
   436→  /// Pinned content-registry version (e.g. "1.0") the letter was issued
   437→  /// under. Immutable after issuance.
   438→  version     String
   439→  /// Printed candidate name, snapshotted from Candidate.fullName at
   440→  /// issuance. Never read from the client at render time.
   441→  printedName String
   442→  /// Official appointment designation, snapshotted at issuance. The
   443→  /// approved MVP value is "Remote Staff" (the designation used by the
   444→  /// official Provisional Letter of Appointment) — not renamed to any
   445→  /// other role title (docs/CONTENT-QUESTIONS.md §21).
   446→  roleTitle   String
   447→  /// Server issuance timestamp. Doubles as the printed letter date for
   448→  /// the MVP; no separate effectiveDate exists (§23).
   449→  issuedAt    DateTime
   450→  createdAt   DateTime  @default(now())
   451→  updatedAt   DateTime  @updatedAt
   452→
   453→  @@map("appointments")
   454→}

   456→/// Server-controlled sequence behind the FFL/HR/PLA/NNN appointment
   457→/// references. A single row per sequence: `value` holds the LAST allocated
   458→/// number; allocation is an atomic `UPDATE ... SET value = value + 1
   459→/// RETURNING value` inside the issuance transaction, so concurrent
   460→/// issuances can never receive the same number (the unique constraint on
   461→/// appointments.appointmentReference stays the final backstop).
   462→///
   463→/// Initialisation: the supplied official sample letter carries
   464→/// FFL/HR/PLA/012 and is treated as the most recently existing appointment
   465→/// (an externally issued document — it is NOT written into the database).
   466→/// The sequence therefore starts at 12, making the first system-issued
   467→/// reference FFL/HR/PLA/013. Documented assumption; see
   468→/// docs/CONTENT-QUESTIONS.md §22.
   469→model AppointmentSequence {
   470→  id    String @id @default(cuid())
   471→  /// Singleton sequence name (one row: "PROVISIONAL_APPOINTMENT").
   472→  name  String @unique
   473→  /// Last allocated sequence number (initialised to 12 by migration).
   474→  value Int
   475→
   476→  @@map("appointment_sequences")
   477→}

   573→/// One emailed email-OTP login code for a Staff Portal account (Stage 7).

   689→/// Transactional email delivery record (outbox pattern, brief §9).
   690→///
   691→/// One row per lifecycle notification INTENT. Rows are created by the
   692→/// central notification service in the SAME database transaction as the
   693→/// meaningful state change they belong to (brief §13), so a committed
   694→/// business event always has a corresponding notification intent even if
   695→/// the email provider is down. The unique `dedupeKey` is the exactly-once
   696→/// latch: user double-clicks, request retries, concurrent requests and
   697→/// route refreshes can never queue a duplicate email (brief §11).
   698→///
   699→/// The actual Mailjet network call happens AFTER the business transaction
   700→/// commits — never while the transaction holds database locks (§16). A
   701→/// Mailjet outage therefore marks rows FAILED/PENDING but can never roll
   702→/// back or corrupt recruitment state (§12/§14).
   703→///
   704→/// No credentials are ever stored here; no rendered HTML is stored — the
   705→/// email body is re-derived from immutable snapshot data at send time so
   706→/// retries stay faithful (staff/appointment/agreement/attempt snapshot
   707→/// rows are immutable by schema design).
   708→model NotificationDelivery {
   709→  id        String             @id @default(cuid())
   710→  /// The candidate the notification belongs to, where applicable. Cascade
   711→  /// delete mirrors every other candidate-owned record.
   712→  candidateId String?
   713→  candidate Candidate?         @relation(fields: [candidateId], references: [id], onDelete: Cascade)
   714→  /// Stable lifecycle key (NotificationType).
   715→  type      NotificationType
   716→  /// Stable idempotency key "<TYPE>:<immutable entity id>" (e.g.
   717→  /// "APPLICATION_RECEIVED:<candidateId>", "ASSESSMENT_PASSED:<attemptId>",
   718→  /// "STAFF_ACTIVATED:<staff row id>"). Unique — the exactly-once latch.
   719→  dedupeKey String             @unique
   720→  /// Recipient email snapshotted at enqueue time (never re-read at send).
   721→  recipientEmail String
   722→  /// Code-owned template registry version used to render the email
   723→  /// (src/server/notifications/notification-templates.ts).
   724→  templateVersion String
   725→  /// Email provider handling delivery (MVP: "mailjet").
   726→  provider  String
   727→  /// Provider message id, when the provider accepted a send.
   728→  providerMessageId String?
   729→  /// PENDING until the provider accepts a send; SENT once accepted
   730→  /// (NOT a claim of inbox delivery — see NotificationStatus).
   731→  status    NotificationStatus @default(PENDING)
   732→  /// Number of delivery attempts made (bounded by config).
   733→  attemptCount Int             @default(0)
   734→  /// When the provider accepted the send (status became SENT).
   735→  sentAt    DateTime?
   736→  /// Last delivery attempt, successful or not.
   737→  lastAttemptAt DateTime?
   738→  /// Safe error code from the last failed attempt (HTTP status class or
   739→  /// NETWORK_ERROR/TIMEOUT). Never contains credentials or bodies.
   740→  lastErrorCode String?
   741→  /// Safe error message from the last failed attempt.
   742→  lastErrorMessage String?
   743→  createdAt DateTime           @default(now())
   744→  updatedAt DateTime           @updatedAt
   745→
   746→  @@index([status])
   747→  @@index([candidateId])
   748→  @@map("notification_deliveries")
